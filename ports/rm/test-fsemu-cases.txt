# test-fsemu-cases.txt -- rm 1.45 FS-UAE test suite
# Category 1 (CLI tool) -- minimum 8 tests required
#
# Flags from source getopt("dfiPRrv"):
#   -d  remove empty directories (dflag)
#   -f  force: ignore nonexistent files, suppress prompts (fflag)
#   -i  interactive prompt (stdin-dependent; skip in automated tests -- stdin not a tty)
#   -P  secure overwrite (accepted but no-op on AmigaOS -- no fstatfs/fsync)
#   -R  recursive removal using fts (rflag)
#   -r  alias for -R (rflag)
#   -v  verbose: print each removed path to stdout (vflag)
#
# Exit codes from source:
#   RC=0   -- RETURN_OK: success
#   RC=1   -- eval=1: error in rm_file/rm_tree (nonexistent file without -f,
#              directory without -d flag, unlink/rmdir failure)
#              NOTE: RC=1 is non-zero but below Amiga WARN threshold (>=5).
#              The test harness supports EXPECT_RC: 1.
#   RC=10  -- usage() -> exit(10): bad args, invalid flag, no args without -f
#             err(10,...): fts_open failure, fts_read errno
#
# IMPORTANT: rm is a destructive filesystem operation. Functional tests use
# ARexx wrapper scripts (test-rm-*.rexx) deployed to WORK:. Each wrapper:
#   1. Creates test files/dirs in T: (Amiga temp volume = RAM:T/)
#   2. Runs WORK:rm with specific arguments
#   3. Verifies result via EXISTS() checks
#   4. Prints "OK" or "FAIL: reason" to stdout
#   5. Cleans up all temp files
#
# Error path tests use direct CMD: lines where the error fires before any
# filesystem change. Error messages go to stderr (not captured by harness),
# so only EXPECT_RC: assertions are used for those tests.
#
# Test count: 17 tests (minimum 8 for Category 1)

# ===========================================================
# ERROR PATH TESTS -- no setup required, error fires immediately
# ===========================================================

EXPECT_RC: 1
