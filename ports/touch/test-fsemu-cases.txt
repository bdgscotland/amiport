# test-fsemu-cases.txt -- touch 1.27 FS-UAE test suite
# Category 1 (CLI tool) -- minimum 8 tests required
#
# Flags from source getopt("acd:fmr:t:"):
#   -a  update access time only (no effect on AmigaOS -- atime not stored)
#   -c  do not create file if nonexistent (creates if exists)
#   -d  ISO 8601 date string [[YYYY-MM-DDThh:mm:ss[.frac][Z]]
#   -f  silently ignored (BSD compatibility stub)
#   -m  update modification time only
#   -r  copy timestamp from reference file
#   -t  numeric timestamp [[CC]YY]MMDDhhmm[.SS]
#
# Exit codes from source:
#   RC=0  -- RETURN_OK: success (file created/updated)
#   RC=10 -- RETURN_ERROR: usage() (no args, invalid flag), bad timestamp,
#             nonexistent -r reference file, file open failure
#
# IMPORTANT: touch produces no stdout output. All operations are silent on
# success; errors go to stderr (not captured by harness). All functional
# tests use ARexx wrapper scripts that:
#   1. Set up preconditions (create/delete temp files in T:)
#   2. Run WORK:touch with specific arguments
#   3. Verify results via ARexx EXISTS() checks
#   4. Print "OK" or "FAIL: reason" to stdout
#   5. Clean up all temp files
#
# Error path tests use direct CMD: lines (error fires before any filesystem
# change; error messages go to stderr only -- use EXPECT_RC: only).
#
# Test count: 18 tests (minimum 8 for Category 1)

# ===========================================================
# ERROR PATH TESTS -- no setup required, error fires immediately
# ===========================================================

# Source: argc == 0 after getopt strip -> usage() -> exit(10)
# Error message goes to stderr (not captured). RC=10 is the gate.
TEST: No arguments prints usage and returns RC 10
CMD: WORK:touch
EXPECT_RC: 10

# Source: getopt default case -> usage() -> exit(10)
# Unknown flag -Z triggers usage error immediately.
TEST: Unknown flag -Z returns usage error RC 10
CMD: WORK:touch -Z T:touch_unused.txt
EXPECT_RC: 10

# Source: stime_arg1 -> goto terr -> errx(10, "out of range...")
# Bad -t format (not digits): errx exits with RC=10 before any file ops.
TEST: Invalid -t format returns RC 10
CMD: WORK:touch -t BADFORMAT T:touch_unused2.txt
EXPECT_RC: 10

# Source: stime_file -> stat() fails -> err(10, fname)
# Nonexistent reference file for -r causes immediate err(10).
TEST: Nonexistent reference file for -r returns RC 10
CMD: WORK:touch -r T:no_such_ref_xyz_001.txt T:touch_unused3.txt
EXPECT_RC: 10

# Source: stime_argd -> goto terr -> errx(10, "out of range...")
# Bad -d format (not ISO 8601): errx exits with RC=10.
TEST: Invalid -d format returns RC 10
CMD: WORK:touch -d not-a-date T:touch_unused4.txt
EXPECT_RC: 10
